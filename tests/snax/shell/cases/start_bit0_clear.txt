# A write to START with bit 0 clear (2) is a plain write, answered at once:
# nothing goes to the unit, so STATUS stays 0 and the counter (never reset
# by an accepted start) reads 0; START reads back 2.
cycles
# a b out  access   result  status
1 1 1      w 1 4    0       0     # 0: JOBS 4
1 1 1      w 0 2    1       0     # 1: START, bit 0 clear: taken at once, no run
1 1 1      r 2      0       0     # 2: STATUS
1 1 1      r 2      0       0     # 3
1 1 1      r 2      0       0     # 4
1 1 1      r 3      0       0     # 5: the counter: no accepted start
1 1 1      r 0      2       0     # 6: START as written
